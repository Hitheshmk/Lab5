#include <iostream>
using namespace std;

int main()
{
    int n;

    cout << "Enter number of vertices: ";
    cin >> n;

    int cost[20][20];

    cout << "Enter Cost Matrix:\n";

    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < n; j++)
        {
            cin >> cost[i][j];
        }
    }

    int visited[20] = {0};
    visited[0] = 1;

    int edges = 0, mincost = 0;

    while (edges < n - 1)
    {
        int min = 999, a = -1, b = -1;

        for (int i = 0; i < n; i++)
        {
            if (visited[i])
            {
                for (int j = 0; j < n; j++)
                {
                    if (!visited[j] && cost[i][j] < min)
                    {
                        min = cost[i][j];
                        a = i;
                        b = j;
                    }
                }
            }
        }

        if (a == -1 || b == -1)
        {
            cout << "Minimum Spanning Tree is not possible.";
            return 0;
        }

        cout << a + 1 << " - " << b + 1
             << " Cost = " << min << endl;

        mincost += min;
        visited[b] = 1;
        edges++;
    }

    cout << "Minimum Cost = " << mincost << endl;

    return 0;
}
